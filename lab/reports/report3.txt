import time, difflib
from functools import lru_cache
from pathlib import Path
import torch
from librosa.core import load
from rapidfuzz.distance import Levenshtein
from quran_transcript import Aya, quran_phonetizer, MoshafAttributes, PhoneticSearch
from quran_muaalem import Muaalem

SR = 16000
LAB = Path(__file__).parent
CONFIGS = [MoshafAttributes(rewaya='hafs', madd_monfasel_len=mf, madd_mottasel_len=mt,
                            madd_mottasel_waqf=mt, madd_aared_len=ar)
           for mf in (4, 5) for mt in (4, 5) for ar in (2, 4, 6)]
BASE = CONFIGS[0]
VOW = set('\u064e\u064f\u0650\u06ea\u0640')
MADD = set('\u0627\u06e6\u06e5')
QAL = '\u0687'
ISTIADHA = 'أَعُوذُ بِٱللَّهِ مِنَ ٱلشَّيْطَـٰنِ ٱلرَّجِيمِ'

def uthmani(ref):
    s, a = map(int, ref.split(':'))
    info = Aya(s, a).get()
    for k in ('uthmani', 'uthmani_script', 'uthmani_text'):
        v = getattr(info, k, None)
        if v:
            return v

@lru_cache(maxsize=None)
def spans_of(text, ci):
    try:
        ph = quran_phonetizer(text, CONFIGS[ci], remove_spaces=True)
    except Exception:
        return tuple()
    words = text.split(' ')
    starts, pos = [], 0
    for w in words:
        starts.append(pos)
        pos += len(w) + 1
    out = []
    for wi, w in enumerate(words):
        lo = hi = None
        for c in range(starts[wi], starts[wi] + len(w)):
            if c >= len(ph.mappings) or ph.mappings[c] is None:
                continue
            a, b = ph.mappings[c].pos
            if b > a:
                lo = a if lo is None else min(lo, a)
                hi = b if hi is None else max(hi, b)
        out.append(ph.phonemes[lo:hi] if lo is not None else '')
    return tuple(out)

def whole(text):
    vs = set()
    for ci in range(len(CONFIGS)):
        sp = spans_of(text, ci)
        if sp:
            vs.add(''.join(sp))
    return sorted(vs)

def units_for(refs):
    units = [{'label': 'الاستعاذة', 'ref': '-', 'n': 0, 'variants': whole(ISTIADHA) + [''], 'optional': True},
             {'label': 'البسملة', 'ref': '-', 'n': 0, 'variants': whole(uthmani('1:1')) + [''], 'optional': True}]
    for r in refs:
        text = uthmani(r)
        words = text.split(' ')
        var = [set() for _ in words]
        for ci in range(len(CONFIGS)):
            full = spans_of(text, ci)
            for k in range(len(words)):
                for sp, idx in ((full, k), (spans_of(' '.join(words[:k + 1]), ci), -1),
                                (spans_of(' '.join(words[k:]), ci), 0), (spans_of(words[k], ci), 0)):
                    if sp:
                        var[k].add(sp[idx])
        for k, w in enumerate(words):
            vs = sorted(v for v in var[k] if v) or ['']
            units.append({'label': w, 'ref': r, 'n': k + 1, 'variants': vs, 'optional': False})
    return units

def align(units, P, K=4, INS=1.0):
    m, n, INF = len(units), len(P), float('inf')
    dp = [[INF] * (n + 1) for _ in range(m + 1)]
    bp = [[None] * (n + 1) for _ in range(m + 1)]
    dp[0][0] = 0
    for j in range(1, n + 1):
        dp[0][j], bp[0][j] = dp[0][j - 1] + INS, ('ins',)
    for i in range(1, m + 1):
        u = units[i - 1]
        lens = {0}
        for v in u['variants']:
            lens.update(range(max(0, len(v) - K), len(v) + K + 1))
        for j in range(n + 1):
            best, b = INF, None
            if j > 0 and dp[i][j - 1] + INS < best:
                best, b = dp[i][j - 1] + INS, ('ins',)
            for l in lens:
                if l > j or dp[i - 1][j - l] >= best:
                    continue
                seg = P[j - l:j]
                c = 0 if (u['optional'] and l == 0) else min(Levenshtein.distance(v, seg) for v in u['variants'])
                if dp[i - 1][j - l] + c < best:
                    best, b = dp[i - 1][j - l] + c, ('unit', l)
            dp[i][j], bp[i][j] = best, b
    i, j, res, ins = m, n, [None] * m, []
    while i > 0 or j > 0:
        b = bp[i][j]
        if b is None:
            break
        if b[0] == 'ins':
            ins.append(j - 1); j -= 1
        else:
            l = b[1]; seg = P[j - l:j]; u = units[i - 1]
            v = min(u['variants'], key=lambda v: Levenshtein.distance(v, seg))
            d = 0 if (u['optional'] and l == 0) else Levenshtein.distance(v, seg)
            res[i - 1] = (seg, v, d); i -= 1; j -= l
    return res, sorted(ins)

def kind(v, seg):
    if not seg:
        return 'كلمة ناقصة'
    strip = lambda s: ''.join(c for c in s if c not in VOW and c not in MADD and c != QAL)
    if strip(v) != strip(seg):
        return 'حروف'
    changed = set()
    for op, a1, a2, b1, b2 in difflib.SequenceMatcher(None, v, seg, autojunk=False).get_opcodes():
        if op != 'equal':
            changed.update(v[a1:a2] + seg[b1:b2])
    if changed and all(c in MADD for c in changed):
        return 'مد'
    if changed and all(c == QAL for c in changed):
        return 'قلقلة'
    return 'حركات'

HIFZ = {'كلمة ناقصة', 'حروف', 'حركات'}
TESTS = [
    ('mine/1.ogg', 'الإخلاص صح', ['112:1', '112:2', '112:3', '112:4']),
    ('mine/2.ogg', 'الإخلاص ناقصة "الصمد"', ['112:1', '112:2', '112:3', '112:4']),
    ('mine/3.ogg', 'الفاتحة 2-4 و"الحمدَ" بالفتح', ['1:2', '1:3', '1:4']),
    ('mine/4.ogg', 'الفاتحة 2-4 صح مع ضوضاء', ['1:2', '1:3', '1:4']),
    ('mine/5.ogg', 'الأعراف 19 مقارنة بالنص الصح', ['7:19']),
    ('mine/5.ogg', 'الأعراف 19 والمطلوب البقرة 35', ['2:35']),
    ('mine/6.ogg', 'الإخلاص بصوت طفل', ['112:1', '112:2', '112:3', '112:4']),
    ('audio/002048.mp3', 'المنشاوي البقرة 48 صح', ['2:48']),
    ('audio/002048.mp3', 'المنشاوي البقرة 48 والمطلوب 123', ['2:123']),
    ('audio/007019.mp3', 'المنشاوي الأعراف 19 والمطلوب البقرة 35', ['2:35']),
]

print('loading model...', flush=True)
model = Muaalem(device='cpu', dtype=torch.float32)

for path, title, refs in TESTS:
    f = LAB / path
    if not f.exists():
        continue
    print('=' * 60)
    print(f'[{path}] {title}  -> {", ".join(refs)}')
    try:
        t0 = time.time()
        units = units_for(refs)
        t_var = time.time() - t0
        wave, _ = load(str(f), sr=SR, mono=True)
        joined = ' '.join(uthmani(r) for r in refs)
        t0 = time.time()
        pred = model([wave], [quran_phonetizer(joined, BASE, remove_spaces=True)], sampling_rate=SR)[0].phonemes.text
        t_inf = time.time() - t0
        t0 = time.time()
        res, ins = align(units, pred)
        t_al = time.time() - t0
        hifz, notes, info = [], [], []
        for u, r in zip(units, res):
            if r is None:
                continue
            seg, v, d = r
            if u['optional']:
                if seg:
                    info.append(u['label'])
                continue
            if d == 0:
                continue
            k = kind(v, seg)
            (hifz if k in HIFZ else notes).append((u, k, v, seg))
        groups, cur = [], []
        for x in ins:
            if cur and x == cur[-1] + 1:
                cur.append(x)
            else:
                if cur: groups.append(cur)
                cur = [x]
        if cur: groups.append(cur)
        extra = [''.join(pred[i] for i in g) for g in groups if len(g) >= 3]
        print(f'audio {len(wave) / SR:.1f}s | variants {t_var:.1f}s | model {t_inf:.1f}s | align {t_al:.1f}s')
        if info:
            print(f'  بدأ بـ: {" + ".join(info)} (مقبول)')
        print(f'  أخطاء حفظ: {len(hifz)}   ملاحظات تجويد: {len(notes)}   كلام زيادة: {len(extra)}')
        for u, k, v, seg in hifz:
            print(f'   ✗ {u["ref"]} كلمة {u["n"]} ({u["label"]}): {k}   الصح="{v}"   اتقال="{seg}"')
        for u, k, v, seg in notes:
            print(f'   • {u["ref"]} كلمة {u["n"]} ({u["label"]}): {k}   الصح="{v}"   اتقال="{seg}"')
        for e in extra:
            print(f'   + زيادة: "{e}"')
    except Exception as ex:
        import traceback; traceback.print_exc()
    print(flush=True)
print('DONE')
